import bpy import math from mathutils import Vector # ========================= # LIMPIAR ESCENA # ========================= bpy.ops.object.select_all(action='SELECT') bpy.ops.object.delete() # ========================= # MATERIALES # ========================= def crear_material(nombre, color, alpha=1.0, roughness=0.35, metallic=0.0): mat = bpy.data.materials.new(nombre) mat.use_nodes = True bsdf = mat.node_tree.nodes.get("Principled BSDF") bsdf.inputs["Base Color"].default_value = color bsdf.inputs["Alpha"].default_value = alpha bsdf.inputs["Roughness"].default_value = roughness bsdf.inputs["Metallic"].default_value = metallic mat.blend_method = 'BLEND' mat.use_screen_refraction = True mat.show_transparent_back = True return mat mat_craneo = crear_material( "Cráneo transparente", (0.85, 0.95, 1.0, 0.22), alpha=0.22, roughness=0.12 ) mat_diente = crear_material( "Diente realista marfil", (0.96, 0.90, 0.76, 1.0), alpha=1.0, roughness=0.28 ) mat_raiz = crear_material( "Raíz dental", (0.78, 0.62, 0.48, 1.0), alpha=1.0, roughness=0.45 ) mat_encia = crear_material( "Encía", (0.75, 0.18, 0.22, 1.0), alpha=1.0, roughness=0.5 ) # ========================= # UTILIDADES # ========================= def agregar_uv_sphere(nombre, ubicacion, escala, material, segmentos=64, anillos=32): bpy.ops.mesh.primitive_uv_sphere_add( segments=segmentos, ring_count=anillos, location=ubicacion ) obj = bpy.context.object obj.name = nombre obj.scale = escala obj.data.materials.append(material) return obj def agregar_cilindro(nombre, ubicacion, radio, profundidad, material, vertices=48): bpy.ops.mesh.primitive_cylinder_add( vertices=vertices, radius=radio, depth=profundidad, location=ubicacion ) obj = bpy.context.object obj.name = nombre obj.data.materials.append(material) return obj def suavizar(obj): bpy.context.view_layer.objects.active = obj obj.select_set(True) bpy.ops.object.shade_smooth() obj.select_set(False) # ========================= # CRÁNEO TRANSPARENTE # ========================= craneo = agregar_uv_sphere( "Cráneo transparente", (0, 0, 1.8), (2.1, 1.45, 2.25), mat_craneo ) suavizar(craneo) # Cavidades orbitarias simuladas ojo_izq = agregar_uv_sphere( "Cavidad ocular izquierda", (-0.72, -1.12, 2.15), (0.38, 0.16, 0.28), mat_craneo ) ojo_der = agregar_uv_sphere( "Cavidad ocular derecha", (0.72, -1.12, 2.15), (0.38, 0.16, 0.28), mat_craneo ) suavizar(ojo_izq) suavizar(ojo_der) # Maxilar superior maxilar = agregar_uv_sphere( "Maxilar superior transparente", (0, -0.15, 0.55), (1.65, 1.0, 0.52), mat_craneo ) suavizar(maxilar) # Mandíbula abierta mandibula = agregar_uv_sphere( "Mandíbula inferior abierta transparente", (0, -0.2, -0.55), (1.7, 1.0, 0.45), mat_craneo ) mandibula.rotation_euler[0] = math.radians(-18) suavizar(mandibula) # ========================= # ENCÍAS # ========================= encia_sup = agregar_uv_sphere( "Encía superior", (0, -0.88, 0.22), (1.55, 0.25, 0.22), mat_encia ) suavizar(encia_sup) encia_inf = agregar_uv_sphere( "Encía inferior", (0, -0.95, -0.72), (1.55, 0.25, 0.22), mat_encia ) encia_inf.rotation_euler[0] = math.radians(-15) suavizar(encia_inf) # ========================= # CREAR DIENTES # ========================= def crear_diente(nombre, x, y, z, escala=(0.16, 0.22, 0.38), superior=True, tipo="incisivo"): # Corona corona = agregar_uv_sphere( nombre + " corona", (x, y, z), escala, mat_diente, segmentos=48, anillos=24 ) suavizar(corona) # Deformación según tipo if tipo == "molar": corona.scale = (escala[0] * 1.35, escala[1] * 1.25, escala[2] * 0.85) elif tipo == "canino": corona.scale = (escala[0] * 0.9, escala[1], escala[2] * 1.25) elif tipo == "premolar": corona.scale = (escala[0] * 1.15, escala[1] * 1.1, escala[2]) # Raíz raiz_z = z - 0.32 if superior else z + 0.32 raiz = agregar_cilindro( nombre + " raíz", (x, y + 0.02, raiz_z), escala[0] * 0.45, 0.48, mat_raiz, vertices=32 ) if superior: raiz.rotation_euler[0] = 0 else: raiz.rotation_euler[0] = math.radians(180) suavizar(raiz) return corona, raiz # Distribución odontograma adulto: 16 superiores + 16 inferiores posiciones = [ (-1.25, "molar"), (-1.05, "molar"), (-0.83, "premolar"), (-0.62, "premolar"), (-0.42, "canino"), (-0.24, "incisivo"), (-0.08, "incisivo"), (0.08, "incisivo"), (0.24, "incisivo"), (0.42, "canino"), (0.62, "premolar"), (0.83, "premolar"), (1.05, "molar"), (1.25, "molar") ] # Dientes superiores for i, (x, tipo) in enumerate(posiciones): y = -1.02 + abs(x) * 0.18 z = 0.02 escala = (0.13, 0.18, 0.32) if tipo == "molar": escala = (0.17, 0.22, 0.28) elif tipo == "premolar": escala = (0.145, 0.20, 0.30) elif tipo == "canino": escala = (0.13, 0.18, 0.38) crear_diente( f"Diente superior {i + 1}", x, y, z, escala=escala, superior=True, tipo=tipo ) # Dientes inferiores for i, (x, tipo) in enumerate(posiciones): y = -1.05 + abs(x) * 0.18 z = -0.98 escala = (0.13, 0.18, 0.32) if tipo == "molar": escala = (0.17, 0.22, 0.28) elif tipo == "premolar": escala = (0.145, 0.20, 0.30) elif tipo == "canino": escala = (0.13, 0.18, 0.38) corona, raiz = crear_diente( f"Diente inferior {i + 1}", x, y, z, escala=escala, superior=False, tipo=tipo ) corona.rotation_euler[0] = math.radians(-15) raiz.rotation_euler[0] = math.radians(165) # ========================= # LUCES # ========================= bpy.ops.object.light_add(type='AREA', location=(0, -4, 5)) luz = bpy.context.object luz.name = "Luz principal suave" luz.data.energy = 700 luz.data.size = 5 bpy.ops.object.light_add(type='POINT', location=(-3, 2, 3)) luz2 = bpy.context.object luz2.name = "Luz lateral" luz2.data.energy = 180 # ========================= # CÁMARA # ========================= bpy.ops.object.camera_add(location=(0, -6.5, 1.1), rotation=(math.radians(82), 0, 0)) bpy.context.scene.camera = bpy.context.object # ========================= # TEXTO / ETIQUETA # ========================= bpy.ops.object.text_add(location=(0, 1.6, -1.4), rotation=(math.radians(90), 0, 0)) texto = bpy.context.object texto.name = "Título odontograma" texto.data.body = "Odontograma 3D - Cráneo transparente" texto.data.size = 0.22 texto.data.align_x = 'CENTER' # ========================= # CONFIGURACIÓN VISUAL # ========================= bpy.context.scene.render.engine = 'CYCLES' bpy.context.scene.cycles.samples = 64 bpy.context.scene.view_settings.view_transform = 'Filmic' bpy.context.scene.view_settings.look = 'Medium High Contrast' bpy.context.scene.view_settings.exposure = 0 bpy.context.scene.view_settings.gamma = 1 # Ajustar origen visual for obj in bpy.context.scene.objects: obj.select_set(False) print("Odontograma 3D generado correctamente.")
